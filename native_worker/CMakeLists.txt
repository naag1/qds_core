cmake_minimum_required(VERSION 3.20)
project(qds_vulkan_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan QUIET)
add_executable(qds_vulkan_worker main.cpp pcap_reader.cpp vulkan_context.cpp vulkan_pipeline.cpp)
if (Vulkan_FOUND)
  target_compile_definitions(qds_vulkan_worker PRIVATE QDS_HAS_VULKAN=1)
  target_link_libraries(qds_vulkan_worker PRIVATE Vulkan::Vulkan)
else()
  message(WARNING "Vulkan SDK not found; building probe-only worker")
endif()
